
import json, math, statistics
import numpy as np
from scipy.optimize import minimize
from pathlib import Path
import sys
repo = Path(r'C:\Users\Drefetr\AppData\Local\Temp\opencode\eq-review-516e')
sys.path.insert(0, str(repo / 'experiments/s04b_refitted_floor'))
import run as rf
cases, _, _ = rf.verify_inputs()
replay, c = rf.load_archive()
geom = c.S.Geometry(r'C:\Users\Drefetr\AppData\Local\Temp\opencode\s04b_floor_native_test\radial.dll')

for reg in (4, 5, 6, 7):
    reg_cases = [cs for cs in cases if cs['regime'] == reg]
    deltas_a_only, deltas_an, n_hats = [], [], []
    for case in reg_cases:
        rows = np.load(rf.safe_input(case['history']))['rows'].copy()
        name = c.G.MATCHED_NAMES[case['regime']-4]
        _, bg, eta, ell, psi = c.G.ROSTER[name]
        law = c.G.Law(name, case['seed'], bg, eta, ell, psi)
        gm = c.G.fitting_metadata(law)
        pars = [law.a, law.d, law.n, math.log(law.c), law.p]
        feats = {}
        for phase, start, end in [('train', 30, 395), ('test', 395, 760)]:
            meta = dict(start=start, end=end, T0=760, beta=gm['beta'], sigma=gm['sigma_km'], bg_floor=gm['bg_floor'], kde_h=gm['kde_h'] or 10)
            arr = c.canonical_arrays(rows[:,0], rows[:,1:3], rows[:,3], np.asarray(gm['rectangles']), np.asarray(gm['reference_pi']), np.asarray(gm['kde_xy']).reshape(-1,2), np.asarray(gm['kde_weight']), meta, pars, geom)
            f = c.S.features(arr, meta, pars[3:5])
            A = rf.integral(law.d, start, end, 760)
            u = np.exp(law.d * np.asarray(f['x'])) * np.asarray(f['b'])
            hp = np.asarray(f['hp'])
            E_trig = float(f['exposure'])
            M = float(f['marks'])
            feats[phase] = (u, hp, A, E_trig, M)
        u_tr, hp_tr, A_tr, H_tr, M_tr = feats['train']
        def nll_bn(x):
            b, n = x
            pt = b * u_tr + n * hp_tr
            if np.any(pt <= 0): return 1e100
            return float(b * A_tr + n * H_tr - np.log(pt).sum() - M_tr)
        res = minimize(nll_bn, [math.exp(law.a), law.n], bounds=[(math.exp(-12), 1.0), (0.0, 0.98)], method='L-BFGS-B', options={'ftol': 1e-14, 'gtol': 1e-12})
        b_hat, n_hat = res.x
        n_hats.append(n_hat)
        u_te, hp_te, A_te, H_te, M_te = feats['test']
        j_old = float(np.log(math.exp(law.a)*u_te + law.n*hp_te).sum() - math.exp(law.a)*A_te - law.n*H_te + M_te)
        j_bn = float(np.log(b_hat*u_te + n_hat*hp_te).sum() - b_hat*A_te - n_hat*H_te + M_te)
        deltas_an.append(j_bn - j_old)
    print(f'S{reg}: mean delta_joint(a,n) = {statistics.mean(deltas_an):.4f}, median n_hat = {statistics.median(n_hats):.4f} [{min(n_hats):.4f}, {max(n_hats):.4f}]')
